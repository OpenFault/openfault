Next.js

# npm i @sentry/nextjs
// instrumentation-client.ts, sentry.server.config.ts, sentry.edge.config.ts

Sentry.init({ dsn: "{{ dsn }}" });

Node

const Sentry = require("@sentry/node");
Sentry.init({ dsn: "{{ dsn }}" });
